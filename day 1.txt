#define TRUE 1

{
    if (NULL)
    {
        printf("NULL");
    }
    else if (FALSE)
    {
        printf("TRUE");
    }
    else
    {
        printf("FALSE");
    }
    return 0;
    
}  

The output is TRUE because NULL is 0 (false), but FALSE is -1, and in C any non-zero value is treated as true.
